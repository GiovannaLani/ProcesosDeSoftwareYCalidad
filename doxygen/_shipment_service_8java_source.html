108 @Scheduled(fixedRate = 60000)
109 public void autoUpdateShipmentStatuses() {

111
112 shipments.forEach(shipment -> {
113 long minutesElapsed = ChronoUnit.MINUTES.between(
114 shipment.getCreatedDate(),
115 LocalDateTime.now()
116 );
117
118 if (shipment.getStatus() == ShipmentStatus.SHIPPED && minutesElapsed >= 1) {
119 shipment.setStatus(ShipmentStatus.IN_TRANSIT);
120 } else if (shipment.getStatus() == ShipmentStatus.IN_TRANSIT && minutesElapsed >= 2) {
121 shipment.setStatus(ShipmentStatus.DELIVERED);
