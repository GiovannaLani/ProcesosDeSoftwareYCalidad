package com.spq.vinted.dto;

import java.util.List;

import com.spq.vinted.model.HomeType;

    }

        super(id, title, description, price,images, isSold);
        this.homeType = type;
    }
    public HomeType getHomeType() {
        return homeType;
    }
    public void setHomeType(HomeType type) {
        this.homeType = type;
    }
}
